# jkit shell hook (fish)
set -gx JKIT_SHELL fish
function __jkit_path
  set -l g '/home/u/.jkit/global'
  for t in sbt gradle maven jdk
    # a *_HOME of your own wins over the global one, so its bin goes on PATH instead
    set -l v (string upper -- $t)_HOME
    test $t = jdk; and set v JAVA_HOME
    set -l h $$v
    # drop a trailing /, so the entry is /x/jdk/bin as jkit writes it, not /x/jdk//bin
    test -n "$h"; and set h (string trim -r -c / -- "$h")
    set -l b $g/$t/bin
    if test -n "$h"; and not contains -- "$h" $g/$t; and test -d "$h/bin"
      set b "$h/bin"
    end
    # first on PATH, also when it is there already but further back: its other places, with or
    # without a trailing /, go
    set -l rest
    for d in $PATH
      test "$d" = "$b"; or test "$d" = "$b/"; or set -a rest "$d"
    end
    set -gx PATH $b $rest
  end
end
__jkit_path
if status is-interactive
  # conf.d files run before config.fish, where a line such as Homebrew's fish_add_path --move
  # can put its bin ahead again: once, at the first prompt, the order is put back
  function __jkit_first_prompt --on-event fish_prompt
    functions -e __jkit_first_prompt
    __jkit_path
  end
end
set -l __jkit_g '/home/u/.jkit/global'
if not set -q JAVA_HOME; and test -e $__jkit_g/jdk; set -gx JAVA_HOME $__jkit_g/jdk; end
if not set -q MAVEN_HOME; and test -e $__jkit_g/maven; set -gx MAVEN_HOME $__jkit_g/maven; end
if not set -q GRADLE_HOME; and test -e $__jkit_g/gradle; set -gx GRADLE_HOME $__jkit_g/gradle; end
if not set -q SBT_HOME; and test -e $__jkit_g/sbt; set -gx SBT_HOME $__jkit_g/sbt; end
if not set -q GRAALVM_HOME; and test -e $__jkit_g/graalvm; set -gx GRAALVM_HOME $__jkit_g/graalvm; end
set -e __jkit_g
set -g __jkit_bin ''
set -g __jkit_root '/home/u/.jkit'
function __jkit_run
  if test -z "$__jkit_bin"; or not test -x "$__jkit_bin"
    # No saved path, or it is gone (a package manager moved jkit): find jkit on PATH,
    # skipping folders under jkit's root, where a tool's bin could hold some other jkit, and
    # relative entries (. or bin), which would run a ./jkit or bin/jkit of the folder cd went to
    set -g __jkit_bin ''
    for d in $PATH
      if string match -q -- '/*' "$d"; and test (string sub -l (string length -- "$__jkit_root/") -- "$d/") != "$__jkit_root/"; and test -x $d/jkit
        set -g __jkit_bin $d/jkit
        break
      end
    end
  end
  if test -n "$__jkit_bin"
    command $__jkit_bin $argv
  else
    echo "jkit: not found on PATH" >&2
    return 127
  end
end
function jkit
  # Ctrl-C can end this function before its rm: a file an earlier run left goes now, or when
  # fish exits. Named as the bash hook names it, so jkit removes it itself on Ctrl-C.
  set -q __jkit_envfile[1]; and rm -f $__jkit_envfile
  set -l tmp /tmp
  test -n "$TMPDIR"; and set tmp (string trim -r -c / -- $TMPDIR)
  set -g __jkit_envfile (mktemp $tmp/jkit-env.XXXXXX); or return 1
  set -l envfile $__jkit_envfile
  set -lx JKIT_ENV_FILE $envfile
  __jkit_run $argv
  set -l rc $status
  test -s $envfile; and source $envfile
  rm -f $envfile
  set -e __jkit_envfile
  return $rc
end
function __jkit_exit --on-event fish_exit
  set -q __jkit_envfile[1]; and rm -f $__jkit_envfile
end
